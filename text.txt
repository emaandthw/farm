
local ISLAND = 107175140535005
local BEAM_MATCH_DIST = 3
local DIG_RETRIES = 3
local SPOT_WAIT = 5
local BEAM_SETTLE = 2
local MIN_DIG_DELAY = 2.05

local RS = game:GetService("ReplicatedStorage")
local LP = game:GetService("Players").LocalPlayer
local TP = game:GetService("TeleportService")
local UIS = game:GetService("UserInputService")

if not game:IsLoaded() then game.Loaded:Wait() end
LP:WaitForChild("PlayerGui")
if not LP.Character then LP.CharacterAdded:Wait() end

local function setupStatusUI()
    local parent = pcall(gethui) and gethui() or game:GetService("CoreGui") or LP:WaitForChild("PlayerGui")
    local old = parent:FindFirstChild("FarmStatusPill")
    if old then old:Destroy() end

    local gui = Instance.new("ScreenGui")
    gui.Name = "FarmStatusPill"
    gui.ResetOnSpawn = false
    gui.DisplayOrder = 10000

    local pill = Instance.new("Frame")
    pill.Name = "Pill"
    pill.AnchorPoint = Vector2.new(0.5, 0)
    pill.Position = UDim2.new(0.5, 0, 0, 18)
    pill.Size = UDim2.new(0, 380, 0, 32)
    pill.BackgroundColor3 = Color3.fromRGB(15, 20, 28)
    pill.BackgroundTransparency = 0.28
    pill.BorderSizePixel = 0
    pill.ClipsDescendants = true
    pill.Parent = gui

    local corner = Instance.new("UICorner", pill)
    corner.CornerRadius = UDim.new(1, 0)

    local stroke = Instance.new("UIStroke", pill)
    stroke.Color = Color3.fromRGB(80, 120, 200)
    stroke.Transparency = 0.35
    stroke.Thickness = 1.6

    local dot = Instance.new("Frame", pill)
    dot.Name = "Dot"
    dot.AnchorPoint = Vector2.new(0, 0.5)
    dot.Position = UDim2.new(0, 12, 0.5, 0)
    dot.Size = UDim2.new(0, 10, 0, 10)
    dot.BackgroundColor3 = Color3.fromRGB(56, 130, 255)
    dot.BorderSizePixel = 0

    local dotCorner = Instance.new("UICorner", dot)
    dotCorner.CornerRadius = UDim.new(1, 0)

    local label = Instance.new("TextLabel", pill)
    label.Name = "StatusText"
    label.AnchorPoint = Vector2.new(0, 0.5)
    label.Position = UDim2.new(0, 30, 0.5, 0)
    label.Size = UDim2.new(1, -40, 1, 0)
    label.BackgroundTransparency = 1
    label.Font = Enum.Font.GothamMedium
    label.TextSize = 13
    label.TextColor3 = Color3.fromRGB(240, 240, 240)
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Text = "Gold Farm | Khoi tao..."
    label.TextTruncate = Enum.TextTruncate.AtEnd

    local dragging, dragStart, startPos
    pill.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = pill.Position
        end
    end)
    pill.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            pill.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    gui.Parent = parent

    return function(text, color)
        label.Text = text
        if color then dot.BackgroundColor3 = color end
    end
end

local updateStatus = setupStatusUI()

local function moveTo(cf)
    (LP.Character or LP.CharacterAdded:Wait()):PivotTo(cf)
end

local function goTo(placeId)
    local id = tonumber(placeId)
    if not id then return end
    updateStatus("Teleport | Chuyen sang " .. tostring(id) .. "...", Color3.fromRGB(180, 100, 255))
    TP:Teleport(id)
end

local function readGold(name)
    local ok, val = pcall(function()
        local hud = LP.PlayerGui:WaitForChild("HUD", 20)
        local gc = hud:WaitForChild("GoldCounters", 20)
        local c = gc:WaitForChild(name, 20):WaitForChild("Counter", 20)
        return tonumber((tostring(c:IsA("ValueBase") and c.Value or c.Text):match("%-?%d+")))
    end)
    return (ok and val) and val or 0
end

local function smelt()
    updateStatus("Island | Cho tai du lieu HUD...", Color3.fromRGB(240, 180, 50))
    local hud = LP.PlayerGui:WaitForChild("HUD", 30)
    local gc = hud and hud:WaitForChild("GoldCounters", 30)
    if not gc then return goTo(_G.FarmPlaceId) end

    local smelter = workspace:WaitForChild("Interactable", 30):WaitForChild("Smelter", 30):WaitForChild("Attachment", 30)
    local prompt = smelter:WaitForChild("@PROMPT", 30)
    prompt.MaxActivationDistance = math.huge
    prompt.RequiresLineOfSight = false

    task.wait(1.5)
    local ore0 = readGold("GoldOreCounter")
    local bar0 = readGold("GoldBarCounter")

    if ore0 <= 0 then
        updateStatus("Island | Khong co quang, quay ve bãi farm...", Color3.fromRGB(255, 100, 100))
        task.wait(1)
        return goTo(_G.FarmPlaceId)
    end

    local t0 = os.clock()
    repeat
        fireproximityprompt(prompt)
        local curOre = readGold("GoldOreCounter")
        updateStatus(string.format("Island | Cho quang vao lo... (Con %d ore)", curOre), Color3.fromRGB(255, 140, 50))
        task.wait(0.8)
    until readGold("GoldOreCounter") <= 0 or os.clock() - t0 > 25

    updateStatus(string.format("Island | Dang nau thoi vang... (Hien co: %d)", bar0), Color3.fromRGB(255, 200, 50))
    t0 = os.clock()
    repeat
        task.wait(0.5)
    until readGold("GoldBarCounter") > bar0 or os.clock() - t0 > 30

    updateStatus(string.format("Island | Nau xong (+%d thoi)! Ve Farm...", math.max(0, readGold("GoldBarCounter") - bar0)), Color3.fromRGB(60, 220, 100))
    task.wait(2)
    goTo(_G.FarmPlaceId)
end

local function farm()
    local WGC = RS:WaitForChild("WGC", 30)
    local Networking = require(WGC.Networking)
    local EconomyConfig = require(WGC.EconomyConfig)

    local function nearestScanPrompt()
        local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if not root then return end
        local best, bestDist
        for _, d in ipairs(workspace:GetDescendants()) do
            if d:IsA("ProximityPrompt") and d.ActionText == "Scan for Gold Ore" and d.Enabled then
                local p = d.Parent
                local pos = p:IsA("Attachment") and p.WorldPosition or p:IsA("BasePart") and p.Position
                if pos then
                    local dist = (pos - root.Position).Magnitude
                    if not bestDist or dist < bestDist then
                        best, bestDist = d, dist
                    end
                end
            end
        end
        return best
    end

    local function findSpotAt(beamPos)
        for _, m in ipairs(workspace:GetChildren()) do
            if m.Name == "Dig Spot" and m:IsA("Model") and (m:GetPivot().Position - beamPos).Magnitude <= BEAM_MATCH_DIST then
                return m
            end
        end
        return nil
    end

    local function addBeamUnique(list, cf)
        for _, existing in ipairs(list) do
            if (existing.Position - cf.Position).Magnitude < 1 then
                return
            end
        end
        table.insert(list, cf)
    end

    local function scanForBeams()
        updateStatus("Farm | Tim tram quet quang...", Color3.fromRGB(60, 140, 255))
        local prompt = nearestScanPrompt()
        if not prompt then return {} end

        local beams = {}
        local conn = workspace.DescendantAdded:Connect(function(x)
            if x:IsA("Model") and x.Name == "Location Beam" then
                task.defer(function()
                    if x.Parent then addBeamUnique(beams, x:GetPivot()) end
                end)
            end
        end)

        updateStatus("Farm | Dang kich hoat quet tia...", Color3.fromRGB(240, 180, 50))
        local t0 = os.clock()
        repeat
            fireproximityprompt(prompt)
            task.wait(0.2)
        until #beams > 0 or os.clock() - t0 > 10

        task.wait(BEAM_SETTLE)

        for _, m in ipairs(workspace:GetChildren()) do
            if m.Name == "Location Beam" and m:IsA("Model") then
                addBeamUnique(beams, m:GetPivot())
            end
        end

        conn:Disconnect()
        updateStatus(string.format("Farm | Tim thay %d tia quang!", #beams), Color3.fromRGB(60, 200, 255))
        return beams
    end

    local count = 0
    local maxCount = nil
    local lastDigTime = 0

    local function digAt(pos)
        local elapsed = os.clock() - lastDigTime
        if elapsed < MIN_DIG_DELAY then
            updateStatus(string.format("Farm | Cho cooldown server (%.1fs)...", MIN_DIG_DELAY - elapsed), Color3.fromRGB(200, 200, 100))
            task.wait(MIN_DIG_DELAY - elapsed)
        end

        for attempt = 1, DIG_RETRIES do
            updateStatus(string.format("Farm | Dang dao quang... (%d/%s)", count, tostring(maxCount or "?")), Color3.fromRGB(56, 130, 255))
            local ok, newCount, level = pcall(function()
                return Networking.C_Dig:InvokeServer(pos)
            end)

            if ok and type(newCount) == "number" and type(level) == "number" then
                lastDigTime = os.clock()
                return newCount, level
            end

            if attempt < DIG_RETRIES then
                task.wait(MIN_DIG_DELAY)
            end
        end
        return nil
    end

    while not maxCount or count < maxCount do
        local beams = scanForBeams()
        if #beams == 0 then
            updateStatus("Farm | Khong tim thay tia, thu lai sau 2s...", Color3.fromRGB(255, 140, 50))
            task.wait(2)
        else
            for idx, cf in ipairs(beams) do
                if maxCount and count >= maxCount then break end
                updateStatus(string.format("Farm | Toi spot %d/%d (Tui: %d/%s)", idx, #beams, count, tostring(maxCount or "?")), Color3.fromRGB(100, 180, 255))
                moveTo(cf)

                local spot
                local s0 = os.clock()
                repeat
                    task.wait(0.1)
                    spot = findSpotAt(cf.Position)
                until spot or os.clock() - s0 > SPOT_WAIT

                if spot then
                    moveTo(spot:GetPivot())
                    local newCount, level = digAt(spot:GetPivot().Position)
                    if newCount then
                        count = newCount
                        maxCount = EconomyConfig.GetLevel(level).InventorySize
                        updateStatus(string.format("Farm | Dao thanh cong! (%d/%d)", count, maxCount), Color3.fromRGB(50, 220, 120))
                    end
                end
            end
        end
    end

    updateStatus("Farm | Day kho! Chuan bi sang Island nau vang...", Color3.fromRGB(255, 100, 220))
    task.wait(2)
    goTo(ISLAND)
end

if game.PlaceId == ISLAND then
    smelt()
else
    farm()
end
